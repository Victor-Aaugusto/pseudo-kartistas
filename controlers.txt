import { Handler } from "express";
import { prisma } from "../database";
import { createCampeonatoSchema, updateCampeonatoSchema } from "../schemas/cmapeonatoSchema";
import { HttpError } from "../error/httpError";
import { ordenaRanking } from "../service/pontuacao";

export class ControlerCampeonato {
    index: Handler = async (req, res, next) =>{
        try {
        const campeonatos = await prisma.campeonato.findMany()
        res.status(200).json({
            data: campeonatos
        })
        } catch (error) {
            next(error)
        }
    }

    create: Handler = async (req, res, next) =>{
        try {
        const data = createCampeonatoSchema.parse(req.body) 

        const newCampeonato = await prisma.campeonato.create({
            data
        })

        if(!newCampeonato) throw new HttpError(400, 'Erro ao criar usuÃ¡rio')

        res.status(201).json({data: newCampeonato})
        } catch (error) {
            next(error)
        }
    }


    show: Handler = async (req, res, next) =>{
        try {
            const campeonato = await prisma.campeonato.findUnique({
            where: {
                id: Number(req.params.id)
            },
            include: {
                CorridaList: true,
                userCamp: {
                    select: {
                        name: true,
                        email: true,
                    }
                }
            }
        })
        res.status(200).json({data: campeonato})
        } catch (error) {
            next(error)
        }
    }

    ranking: Handler = async (req, res, next) =>{
        try {
            const campeonato = await prisma.campeonato.findUnique({
            where: {
                id: Number(req.params.id)
            },
            include: {
                CorridaList: {
                   include:{
                    resultadoCorridas: {
                        include:{
                            user:{
                                select: {
                                name: true,
                                email: true
                            }
                        }
                        }
                    }                    
                   }
                }
            }
        })

        if(!campeonato) throw new HttpError(404, 'Campeonato nÃ£o localizado')

        const ranking = ordenaRanking(campeonato)

        res.status(200).json({data: ranking})
        } catch (error) {
            next(error)
        }
}

    update: Handler = async (req, res, next) =>{
        try {
            const campeonatoExistis = await prisma.campeonato.findUnique({
            where: {
                id: Number(req.params.id)
            }
          })

          if(!campeonatoExistis) throw new HttpError(404, 'Campeonato nÃ£o existe')

          const data = updateCampeonatoSchema.parse(req.body) 
          const updatedCampeonato = await prisma.campeonato.update({
            data: data,
            where: {
                id: Number(req.params.id)
            }
          })

          res.status(200).json({data:updatedCampeonato})
        } catch (error) {
            next(error)
        }
    }


    delete: Handler = async (req, res, next) =>{
        try {
          const campeonatoExistis = await prisma.campeonato.findUnique({
            where: {
                id: Number(req.params.id)
            }
          })

          if(!campeonatoExistis) throw new HttpError(404, 'Campeonato nÃ£o existe')

          await prisma.campeonato.deleteMany({
            where: {
                id: Number(req.params.id)
            }
          })

          res.status(200).json({message: 'Campeonato excluido com sucesso!'})
        } catch (error) {
            next(error)
        }
    }
}

import { Handler } from "express";
import { prisma } from "../database";
import { createResultManySchema, createResultSchema, updatedResultManySchema, updatedResultSchema } from "../schemas/resultadoSchema";
import { HttpError } from "../error/httpError";
import { calculaPontos } from "../service/pontuacao";
import { Prisma } from "../../generated/prisma/client";
import { campExiste, corridaDuplicate, corridaExist } from "../service/validade";
import { time, timeEnd } from "node:console";

export class ControlerResultado {
    index: Handler = async(req, res, next) => {
        try {
            const resultadoCorrida = await prisma.resultadoCorrida.findMany()
            res.status(200).json({data: resultadoCorrida})
        } catch (error) {
            next(error)
        }
    }

    create: Handler = async(req, res, next) => {
        try {
            const data = createResultSchema.parse(req.body)
            const points = calculaPontos(data.posicao)

            const newResult = await prisma.resultadoCorrida.create({
                data: {...data, pontos: points}
            })

            res.status(201).json({data: newResult})
        } catch (error) {
            next(error)
        }
    }

    createMany: Handler = async (req, res, next) =>{
        const data = createResultManySchema.parse(req.body)

        if(!data) throw new HttpError(404, 'Resultados nÃ£o encontrados')

        const existisCorrida = await prisma.corridas.findFirst({
            where: {
                id: Number(req.params.corridaId),
            }
        })
        
        data.every(user => {
            if(user.corridaId !== Number(req.params.corridaId)){
                throw new HttpError(400, 'Dados inconsistentes!')
            } else {
                return true
            }
        })

        if(!existisCorrida) throw new HttpError(404, 'Corrida nÃ£o encontrada')

        const existisCamp = await prisma.campeonato.findFirst({
            where: {
                id: Number(req.params.campeonatoId)
            }
        })

        const existisCampECorddia = existisCorrida?.campeonatoId === existisCamp?.id

        if(!existisCampECorddia) throw new HttpError(404, 'Campeonato nÃ£o encontrado')

        const resultado = data.map(user =>( {
            ...user,
            pontos: calculaPontos(user.posicao)
            })
        )

        const newResults = await prisma.resultadoCorrida.createMany({
            data: resultado
        })

        res.status(201).json({data: newResults})
    }

    show: Handler = async(req, res, next) => {
        try {
            const resultado = await prisma.resultadoCorrida.findUnique({
                where:{
                    corridaId_userId: {
                        userId: Number(req.params.userId),
                        corridaId: Number(req.params.corridaId)
                    }
                }
            })

             if(!resultado) throw new HttpError(404, 'NÃ£o localizado')

            res.status(200).json({data: resultado})
        } catch (error) {
            next(error)
        }
    }

    update: Handler = async(req, res, next) => {
        try {
            const resultadoExists = await prisma.resultadoCorrida.findUnique({
                where: {
                corridaId_userId: {
                    corridaId: Number(req.params.corridaId),
                    userId: Number(req.params.userId)
                }
            }
        })


            if(!resultadoExists) throw new HttpError(404, 'NÃ£o localizado')

            const data = updatedResultSchema.parse(req.body)


            const points = data.posicao ?? resultadoExists.posicao

            const updatedPoints = calculaPontos(points)


            const updatedResultado = await prisma.resultadoCorrida.update({
                data: {...data, pontos: updatedPoints},
                where:{
                    corridaId_userId: {
                        userId: Number(req.params.userId),
                        corridaId: Number(req.params.corridaId)
                    }
                }
            })

            res.status(200).json({data: updatedResultado})
        } catch (error) {
            next(error)
        }
    }


    updateMany: Handler = async(req, res, next) =>{
        try{
        const data = updatedResultManySchema.parse(req.body)

        if(!data) throw new HttpError(404, 'Resultados nÃ£o encontrados')

        const existisCorrida = await corridaExist(Number(req.params.corridaId))
        
        if(!existisCorrida) throw new HttpError(404, 'Corrida nÃ£o encontrada')

        const corridaDuplicat = await corridaDuplicate(data, Number(req.params.corridaId))

        if(!corridaDuplicat) throw new HttpError(404, 'Parametros incorretos')

        const existisCamp = await campExiste(Number(req.params.campeonatoId))

        const existisCampECorddia = existisCamp?.camp.id === existisCorrida.existisCorrida.campeonatoId

        if(!existisCampECorddia) throw new HttpError(404, 'Campeonato nÃ£o encontrado')

        const dadosSeparadosId = data.flatMap(dados => dados.userId)
        const dadosSeparadosPosicao = data.flatMap(dados => dados.posicao)
        const idRepeat = new Set(dadosSeparadosId).size !== dadosSeparadosId.length
        const posicaoRepeat = new Set(dadosSeparadosPosicao).size !== dadosSeparadosPosicao.length

        if(idRepeat || posicaoRepeat) throw new HttpError(400, 'requisiÃ§Ã£o contÃ©m pilotos/posiÃ§Ãµes duplicados')


        const resultado = data.map(result =>( {
            ...result,
            pontos: calculaPontos(result.posicao)
        
        }))


        const newResult = await prisma.$transaction(async (tx) =>{
            for (const user of resultado){
              
              await  tx.resultadoCorrida.update({
                    where:{
                        corridaId_userId:{
                            userId: user.userId,
                            corridaId: user.corridaId
                        }
                    },
                    data: {
                        posicao: user.posicao + 10,
                        pontos: 0
                    }
                })
                
            }

            let resultadoFinal:any = []

             for (const user of resultado){
              
             const result = await  tx.resultadoCorrida.update({
                    where:{
                        corridaId_userId:{
                            userId: user.userId,
                            corridaId: user.corridaId
                        }
                    },
                    data: {
                        posicao: user.posicao,
                        pontos: user.pontos
                    }
                })
                resultadoFinal.push(result)
                
            }
            return resultadoFinal
        }

    )

        res.status(201).json({data: newResult})
    } catch(error){
        next(error)
    }
    }

    delete: Handler = async(req, res, next) => {
        try {
            const resultadoExists = await prisma.resultadoCorrida.findUnique({
                where: {
                    corridaId_userId: {
                        userId: Number(req.params.userId),
                        corridaId: Number(req.params.corridaId)
                    }
                }
            })

            if(!resultadoExists) throw new HttpError(404, 'NÃ£o localizado')

            await prisma.resultadoCorrida.delete({
                where: {
                  corridaId_userId: {
                        userId: Number(req.params.userId),
                        corridaId: Number(req.params.corridaId)
                    }
                }
            })

            res.status(200).json({message: "Resultado deletado com sucesso"})
        } catch (error) {
            next(error)
        }
    }
}

===== userControler.ts =====
import { Handler } from "express";
import { prisma } from "../database";
import { createUserSchema, loginUserSchema, UpdateUserSchema } from "../schemas/userSchema";
import { HttpError } from "../error/httpError";
import { email } from "zod";

const jwt = require('jsonwebtoken')
const bcrypt = require('bcrypt')

export class ControlerUser {

    login: Handler = async (req, res, next) =>{
        try {
            const data = loginUserSchema.parse(req.body)

        const user = await prisma.user.findUnique({
            where: {
                email: data.email
            }
        })

        if(!user) throw new HttpError(404, 'Usuario nÃ£o localizado')

        const validSenha = await bcrypt.compare(req.body.senha, user.senha)

        if(!validSenha) throw new HttpError(404, 'Senha invÃ¡lida')
           
            
        const payload = {email: user.email, nome: user.name, id: user.id}
        const assinatura = jwt.sign(payload, process.env.SECRETKEY, {expiresIn: '1h'})
        res.json({data: assinatura})
        } catch (error) {
            next(error)
        }
    } 

    index: Handler = async (req, res, next) => {
        try {
            const users = await prisma.user.findMany()

            const dataUsers = users.map(user => {
                return {nome: user.name, email: user.email}
            })

            res.status(200).json({data: dataUsers})
        } catch (error) {
            next(error)
        }
    }

    create: Handler = async (req, res, next) =>{
       try {
        const data = createUserSchema.parse(req.body)

        const pass = await bcrypt.hash(data.senha, 10)

        const User ={...data, senha: pass}

        const newUser = await prisma.user.create({
            data: User
        })

        if(!User) throw new HttpError(400,'Falha ao criar usuÃ¡rio')


        const payload = {name: newUser.name, email: newUser.email}

        res.status(201).json({data: payload})
       } catch (error) {
            next(error)
       }

    }

    show: Handler = async (req, res, next) => {
        try {
            const user = await prisma.user.findUnique({
                where: {
                    id: Number(req.params.id)
                },
                select: {
                    name: true,
                    email: true,
                    include: {
                     campeonato: true,
                     corridas: true
                    }
                }
            })

            if(!user) throw new HttpError(404,'usuario nÃ£o localizado')

            const payload = {name: user.name, email: user.email}

            res.status(200).json({data: payload})
        } catch (error) {
            next(error)
        }
    }



    update: Handler = async (req, res, next) => {
        try {
            const userExist = await prisma.user.findUnique({
            where: {
                id: Number(req.params.id)
            }
           })

            if(!userExist) throw new HttpError(404,'usuario nÃ£o localizado')

            const data = UpdateUserSchema.parse(req.body)

            const updatedUser = await prisma.user.update({
                data: data,
                where: {
                    id: Number(req.params.id)
                }
            })

            const payload = {name: updatedUser.name, email: updatedUser.email, role: updatedUser.role}

            res.status(200).json({data: payload})
        } catch (error) {
            next(error)
        }
    }


    delete: Handler = async (req, res, next) => {
        try {
           const userExist = await prisma.user.findUnique({
            where: {
                id: Number(req.params.id)
            }
           })

           if(!userExist) throw new HttpError(404,'usuario nÃ£o localizado')

            await prisma.user.delete({
                where: {
                    id: Number(req.params.id)
                }
            })

            res.status(200).json({message: "Usuario deletado com suceso!"})
        } catch (error) {
            next(error)
        }
    }
}
